1. Project Overview & System Architecture

The Appointment Scheduling System (Schedulr) is an enterprise-grade full-stack web application designed to allow clients to discover services, view real-time time-zone aware availability, book appointments, and track their lifecycle. It provides role-based portals for Clients, Service Providers, and Administrators.

Key Functional Modules

  • Client Booking Flow: 4-step wizard (Service → Provider → Live Slot → Confirmation) with zero double-booking concurrency guarantees.
  • Dynamic Slot Engine: Computes bookable windows mathematically by intersecting provider operating hours, subtracting blocked time-off, active bookings, buffer times, and historical timestamps.
  • Provider Schedule Manager: Weekly recurring hours editor (Monday–Sunday), ad-hoc date/time blockouts, and status progression (Confirm, Complete, No-Show, Cancel).
  • Administrative Operations: Service catalog management, provider onboarding, utilization rates, and audit logs.
  • Calendar Synchronization: Generates RFC 5545 .ics calendar files downloadable directly into Google Calendar, Outlook, and Apple Calendar.

3. Core Algorithm: Slot Availability Engine

The core scheduling algorithm that calculates dynamic free slots while subtracting appointments, buffers, and time-off:

// server/src/lib/slots.ts

export function computeAvailableSlots(params: ComputeSlotsParams): ComputedSlot[] {
  const {
    dateStr, clientTimezone, providerTimezone,
    availabilityRules, blockedSlots, existingAppointments,
    durationMinutes, bufferMinutes = 0, nowUtc = new Date()
  } = params;

  const clientZone = isValidTimezone(clientTimezone) ? clientTimezone : 'UTC';
  const providerZone = isValidTimezone(providerTimezone) ? providerTimezone : 'UTC';

  const clientDayStart = DateTime.fromISO(dateStr, { zone: clientZone }).startOf('day');
  const clientDayEnd = clientDayStart.endOf('day');
  const dayStartMs = clientDayStart.toMillis();
  const dayEndMs = clientDayEnd.toMillis();
  const durationMs = durationMinutes * 60_000;
  const requestedBufferMs = Math.max(0, bufferMinutes) * 60_000;
  const stepMs = (durationMinutes >= 30 ? 30 : durationMinutes) * 60_000;

  // Build busy intervals from blocked time-off + existing active appointments + buffers
  const busyIntervals: Array<{ startMs: number; endMs: number }> = [];
  
  for (const block of blockedSlots) {
    busyIntervals.push({ startMs: block.startUtc.getTime(), endMs: block.endUtc.getTime() });
  }
  
  for (const appt of existingAppointments) {
    const apptBufferMs = Math.max(0, appt.bufferMinutes ?? 0) * 60_000;
    busyIntervals.push({
      startMs: appt.startUtc.getTime() - requestedBufferMs,
      endMs: appt.endUtc.getTime() + Math.max(apptBufferMs, requestedBufferMs)
    });
  }

  const slots: ComputedSlot[] = [];
  const rulesByDay = new Map(availabilityRules.filter(r => r.isEnabled).map(r => [r.dayOfWeek, r]));

  let cursorDate = clientDayStart.setZone(providerZone).startOf('day');
  const providerEndDate = clientDayEnd.setZone(providerZone).startOf('day');

  while (cursorDate <= providerEndDate) {
    const rule = rulesByDay.get(cursorDate.weekday % 7);
    if (rule) {
      const [sH, sM] = rule.startTime.split(':').map(Number);
      const [eH, eM] = rule.endTime.split(':').map(Number);
      const windowStart = cursorDate.set({ hour: sH, minute: sM, second: 0 }).toMillis();
      const windowEnd = cursorDate.set({ hour: eH, minute: eM, second: 0 }).toMillis();

      let slotStartMs = windowStart;
      while (slotStartMs + durationMs <= windowEnd) {
        const slotEndMs = slotStartMs + durationMs;
        const withinDay = slotStartMs >= dayStartMs && slotStartMs <= dayEndMs;
        const inFuture = slotStartMs > nowUtc.getTime();

        if (withinDay && inFuture) {
          const hasConflict = busyIntervals.some(b => 
            slotStartMs < b.endMs && slotEndMs > b.startMs
          );

          if (!hasConflict) {
            const startDt = DateTime.fromMillis(slotStartMs, { zone: clientZone });
            slots.push({
              startUtc: startDt.toUTC().toISO()!,
              endUtc: DateTime.fromMillis(slotEndMs, { zone: clientZone }).toUTC().toISO()!,
              label: startDt.toFormat('h:mm a')
            });
          }
        }
        slotStartMs += stepMs;
      }
    }
    cursorDate = cursorDate.plus({ days: 1 });
  }
  return slots;
}

4. Concurrency & Overlap Prevention (Prisma Transaction)

Guarantees atomic slot reservation and prevents race-condition double bookings:

// server/src/routes/appointments.ts

router.post('/', requireAuth, validateBody(createAppointmentSchema), asyncHandler(async (req, res) => {
  const { providerId, serviceId, startUtc: startUtcStr, notes } = req.body;
  const startUtc = new Date(startUtcStr);

  const created = await prisma.$transaction(async (tx) => {
    // 1. Re-validate slot availability inside the atomic transaction lock
    const { provider, service, endUtc } = await assertSlotBookableInTx(tx, {
      providerId, serviceId, startUtc
    });

    // 2. Generate unique human-readable tracking code
    const referenceCode = `APT-${crypto.randomBytes(3).toString('hex').toUpperCase()}`;

    // 3. Insert confirmed appointment
    const appointment = await tx.appointment.create({
      data: {
        referenceCode,
        clientId: req.user!.userId,
        providerId,
        serviceId,
        startUtc,
        endUtc,
        status: 'CONFIRMED',
        notes: notes || '',
        events: {
          create: {
            actorId: req.user!.userId,
            toStatus: 'CONFIRMED',
            note: 'Appointment booked successfully'
          }
        }
      },
      include: appointmentInclude
    });

    // 4. Create in-app notification alerts for both client and provider
    await tx.notification.createMany({
      data: [
        {
          userId: req.user!.userId,
          appointmentId: appointment.id,
          type: 'BOOKING_CONFIRMED',
          message: `Booking ${referenceCode} confirmed for ${service.name}.`
        },
        {
          userId: provider.userId,
          appointmentId: appointment.id,
          type: 'BOOKING_CONFIRMED',
          message: `New booking ${referenceCode} received from ${req.user!.email}.`
        }
      ]
    });

    return appointment;
  });

  res.status(201).json({ appointment: formatAppointment(created) });
}));

5. Authentication & Role-Based Middleware (RBAC)

Secure JWT-based session verification with role guard:

// server/src/middleware/auth.ts

export function requireAuth(req: Request, _res: Response, next: NextFunction): void {
  const token = req.cookies?.schedulr_token || req.headers.authorization?.replace('Bearer ', '');
  if (!token) {
    return next(new AppError('Authentication required', 401, 'UNAUTHORIZED'));
  }
  try {
    req.user = verifyToken(token); // Decodes { userId, role, email }
    next();
  } catch {
    next(new AppError('Invalid or expired session', 401, 'UNAUTHORIZED'));
  }
}

export function requireRole(...roles: Array<'CLIENT' | 'PROVIDER' | 'ADMIN'>) {
  return (req: Request, _res: Response, next: NextFunction): void => {
    if (!req.user) return next(new AppError('Authentication required', 401));
    if (!roles.includes(req.user.role)) {
      return next(new AppError('Forbidden: Insufficient privileges', 403));
    }
    next();
  };
}

// client/src/pages/BookAppointmentPage.tsx (Excerpt)

export function BookAppointmentPage() {
  const [services, setServices] = useState<Service[]>([]);
  const [selectedService, setSelectedService] = useState<Service | null>(null);
  const [selectedProvider, setSelectedProvider] = useState<Provider | null>(null);
  const [selectedDate, setSelectedDate] = useState<string>(getTodayIsoDate());
  const [slots, setSlots] = useState<TimeSlot[]>([]);
  const [selectedSlot, setSelectedSlot] = useState<TimeSlot | null>(null);

  // Load real-time slots whenever provider, service, or date changes
  useEffect(() => {
    if (!selectedService || !selectedProvider) return;
    api.slots.list({
      providerId: selectedProvider.id,
      serviceId: selectedService.id,
      date: selectedDate,
      tz: Intl.DateTimeFormat().resolvedOptions().timeZone
    }).then(res => setSlots(res.slots));
  }, [selectedService, selectedProvider, selectedDate]);

  const handleBooking = async () => {
    if (!selectedSlot) return;
    const res = await api.appointments.create({
      providerId: selectedProvider!.id,
      serviceId: selectedService!.id,
      startUtc: selectedSlot.startUtc
    });
    alert(`Booked successfully! Reference: ${res.appointment.referenceCode}`);
  };

  return (
    <div className="max-w-4xl mx-auto p-6 space-y-6">
      <h1 className="text-2xl font-bold">Book an Appointment</h1>
      {/* 1. Date Picker */}
      <input type="date" value={selectedDate} onChange={e => setSelectedDate(e.target.value)} />
      
      {/* 2. Slot Grid */}
      <div className="grid grid-cols-4 gap-2">
        {slots.map(s => (
          <button key={s.startUtc} onClick={() => setSelectedSlot(s)}
            className={selectedSlot?.startUtc === s.startUtc ? "bg-indigo-600 text-white" : "border p-2"}>
            {s.label}
          </button>
        ))}
      </div>

      <button onClick={handleBooking} disabled={!selectedSlot} className="bg-indigo-600 text-white px-6 py-2 rounded">
        Confirm Booking
      </button>
    </div>
  );
}
